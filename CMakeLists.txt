cmake_minimum_required(VERSION 3.20)
project(JetCash VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(JETCASH_RANDOMX "Fetch and build the pinned RandomX backend" ON)
option(JETCASH_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)

add_library(jetcash_core src/consensus.cpp src/pow.cpp)
target_include_directories(jetcash_core PUBLIC include)
if(MSVC)
  target_compile_options(jetcash_core PRIVATE /W4 /WX)
else()
  target_compile_options(jetcash_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(JETCASH_SANITIZERS AND NOT MSVC)
  target_compile_options(jetcash_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(jetcash_core PUBLIC -fsanitize=address,undefined)
endif()
if(JETCASH_RANDOMX)
  include(FetchContent)
  FetchContent_Declare(randomx
    GIT_REPOSITORY https://github.com/tevador/RandomX.git
    GIT_TAG 12f2c2ffe2108d6cf54c391fee33c8bc3646cdab)
  FetchContent_MakeAvailable(randomx)
  target_link_libraries(jetcash_core PRIVATE randomx)
  target_include_directories(jetcash_core PRIVATE ${randomx_SOURCE_DIR}/src)
  target_compile_definitions(jetcash_core PRIVATE JETCASH_HAS_RANDOMX=1)
endif()
add_executable(jetcash src/main.cpp)
set_target_properties(jetcash PROPERTIES OUTPUT_NAME jetcash-policy)
target_link_libraries(jetcash PRIVATE jetcash_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(jetcash_tests tests/core_tests.cpp)
  target_link_libraries(jetcash_tests PRIVATE jetcash_core)
  add_test(NAME jetcash_core COMMAND jetcash_tests)
  if(JETCASH_RANDOMX)
    add_executable(jetcash_randomx_vector tests/randomx_vector.cpp)
    target_include_directories(jetcash_randomx_vector PRIVATE ${randomx_SOURCE_DIR}/src)
    target_link_libraries(jetcash_randomx_vector PRIVATE jetcash_core randomx)
    add_test(NAME randomx_upstream_vector COMMAND jetcash_randomx_vector)
  endif()
endif()
